**Partials rule:** every page has

```html
<div data-include="/<app>/header.html"></div>   <!-- app-level -->
<div data-include="/<app>/<module>/header.html"></div> <!-- optional module-level -->
...page content...
<div data-include="/<app>/<module>/footer.html"></div>
<div data-include="/<app>/footer.html"></div>
```

`include.js` fetches each, injects in place, and fires `partials:ready`. `i18n.js` waits for that event before translating `data-i18n` — so partials can contain their own `data-i18n` keys.

**Per-page translation rule:**

```html
<script>
  window.I18N = {
    'page.title':     { en: 'Lesson 1', fr: 'Leçon 1', ar: 'الدرس الأول' },
    'lesson.start':   { en: 'Start',    fr: 'Commencer', ar: 'ابدأ' }
  };
</script>
<h1 data-i18n="page.title"></h1>
<button data-i18n="lesson.start"></button>
```

Keys are page-local; missing keys fall back to the element's current textContent (so an un-translated key never blanks the UI).

## `shared/store.js`

```js
// shared/store.js — storage facade. Async from day one so Supabase is a drop-in swap.
import { CONFIG } from './config.js';

const PREF_KEYS = { mode: 'edu:mode', lang: 'edu:lang', profile: 'edu:profile' };

const mirror = {
  read(k, d) { try { const v = localStorage.getItem(k); return v == null ? d : JSON.parse(v); } catch { return d; } },
  write(k, v) { localStorage.setItem(k, JSON.stringify(v)); },
  remove(k) { localStorage.removeItem(k); },
  keys(prefix) {
    const out = [];
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k && k.startsWith(prefix)) out.push(k);
    }
    return out;
  }
};

const local = {
  async pushPrefs() {},
  async pullPrefs() { return null; },
  async getApp(appId, key) { return mirror.read(`${appId}:${key}`, null); },
  async setApp(appId, key, val) { mirror.write(`${appId}:${key}`, val); },
  async removeApp(appId, key) { mirror.remove(`${appId}:${key}`); },
  async clearApp(appId) { mirror.keys(`${appId}:`).forEach((k) => mirror.remove(k)); },
  async setProfile(p) { mirror.write(PREF_KEYS.profile, p); }
};

const supabase = {
  async pushPrefs() {},
  async pullPrefs() { return null; },
  async getApp()     { throw new Error('supabase backend not wired yet'); },
  async setApp()     { throw new Error('supabase backend not wired yet'); },
  async removeApp()  { throw new Error('supabase backend not wired yet'); },
  async clearApp()   { throw new Error('supabase backend not wired yet'); },
  async setProfile(p) { mirror.write(PREF_KEYS.profile, p); }
};

const backend = CONFIG.backend === 'supabase' ? supabase : local;

export const prefs = {
  /** Sync read of local mirror. Used pre-paint to avoid flash. */
  get() {
    return {
      mode:    mirror.read(PREF_KEYS.mode, 'light'),
      lang:    mirror.read(PREF_KEYS.lang, null),
      profile: mirror.read(PREF_KEYS.profile, null)
    };
  },
  async setMode(mode) {
    mirror.write(PREF_KEYS.mode, mode);
    window.dispatchEvent(new CustomEvent('edu:mode-changed', { detail: { mode } }));
    await backend.pushPrefs();
  },
  async setLang(lang) {
    mirror.write(PREF_KEYS.lang, lang);
    window.dispatchEvent(new CustomEvent('edu:lang-changed', { detail: { lang } }));
    await backend.pushPrefs();
  },
  async setProfile(profile) {
    mirror.write(PREF_KEYS.profile, profile);
    await backend.setProfile(profile);
  },
  async sync() {
    const remote = await backend.pullPrefs();
    if (!remote) return;
    if (remote.mode)    mirror.write(PREF_KEYS.mode, remote.mode);
    if (remote.lang)    mirror.write(PREF_KEYS.lang, remote.lang);
    if (remote.profile) mirror.write(PREF_KEYS.profile, remote.profile);
  }
};

export const appData = {
  async get(appId, key, def = null) {
    if (CONFIG.backend === 'local') return mirror.read(`${appId}:${key}`, def);
    const cached = mirror.read(`${appId}:${key}`, undefined);
    if (cached !== undefined) return cached;
    try {
      const v = await backend.getApp(appId, key);
      if (v !== null) mirror.write(`${appId}:${key}`, v);
      return v ?? def;
    } catch { return def; }
  },
  async set(appId, key, val) {
    mirror.write(`${appId}:${key}`, val);
    await backend.setApp(appId, key, val);
  },
  async remove(appId, key) {
    mirror.remove(`${appId}:${key}`);
    await backend.removeApp(appId, key);
  },
  async clear(appId) {
    mirror.keys(`${appId}:`).forEach((k) => mirror.remove(k));
    await backend.clearApp(appId);
  }
};
```

## `shared/include.js`

Loads `[data-include]` partials — top-level in parallel, nested recursively — then fires one `partials:ready` event that `i18n.js` waits for.

```js
// shared/include.js — <div data-include="/arabic/header.html"></div> loader.
const cache = new Map();

async function fetchPartial(url) {
  if (cache.has(url)) return cache.get(url);
  const res = await fetch(url, { cache: 'no-cache' });
  if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`);
  const text = await res.text();
  cache.set(url, text);
  return text;
}

async function processNode(node) {
  if (!node.isConnected) return;                // already replaced by another parent
  const url = node.dataset.include;
  if (!url) return;
  try {
    const html = await fetchPartial(url);
    const tpl = document.createElement('template');
    tpl.innerHTML = html.trim();
    const fragment = tpl.content;
    const nested = [...fragment.querySelectorAll('[data-include]')];
    node.replaceWith(fragment);
    await Promise.all(nested.map(processNode));
  } catch (err) {
    console.error('[include]', url, err);
    node.replaceWith(document.createComment(` include failed: ${url} `));
  }
}

async function run() {
  const roots = [...document.querySelectorAll('[data-include]')];
  await Promise.all(roots.map(processNode));
  document.dispatchEvent(new CustomEvent('partials:ready'));
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', run);
} else {
  run();
}
```

## `shared/i18n.js`

```js
// shared/i18n.js — applies window.I18N[key][lang] to [data-i18n] and [data-i18n-attr].
import { prefs } from './store.js';
import { CONFIG } from './config.js';

function currentLang() {
  return prefs.get().lang || document.documentElement.lang || CONFIG.defaultLang;
}

function translateNode(el, lang) {
  const key = el.dataset.i18n;
  if (!key) return;
  const entry = (window.I18N || {})[key];
  if (!entry) return;
  const val = entry[lang] ?? entry.en;
  if (val != null) el.textContent = val;
}

function translateAttr(el, lang) {
  const spec = el.dataset.i18nAttr;
  if (!spec) return;
  const dict = window.I18N || {};
  // format: "attr:key;attr2:key2"
  for (const pair of spec.split(';')) {
    const [attr, key] = pair.split(':').map((s) => s.trim());
    if (!attr || !key) continue;
    const entry = dict[key];
    if (!entry) continue;
    const val = entry[lang] ?? entry.en;
    if (val != null) el.setAttribute(attr, val);
  }
}

export function applyI18n(root = document) {
  const lang = currentLang();
  if (root === document || root === document.documentElement) {
    document.documentElement.lang = lang;
    document.documentElement.dir  = lang === 'ar' ? 'rtl' : 'ltr';
  }
  if (root.matches?.('[data-i18n]'))      translateNode(root, lang);
  if (root.matches?.('[data-i18n-attr]')) translateAttr(root, lang);
  root.querySelectorAll?.('[data-i18n]').forEach((el) => translateNode(el, lang));
  root.querySelectorAll?.('[data-i18n-attr]').forEach((el) => translateAttr(el, lang));
}

// Re-apply on events that can change what's on screen.
document.addEventListener('partials:ready',  () => applyI18n());
window.addEventListener('edu:lang-changed',   () => applyI18n());
window.addEventListener('storage', (e) => { if (e.key === 'edu:lang') applyI18n(); });

// Always translate once after the DOM settles.
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', () => applyI18n());
} else {
  applyI18n();
}
```

// shared/ui.js — header controls: theme toggle, language select, user chip.
import { prefs } from './store.js';
import { auth }  from './auth.js';
import { applyI18n } from './i18n.js';

function currentLang() { return prefs.get().lang || document.documentElement.lang || 'en'; }

export function initTheme() {
  const btn = document.getElementById('theme-btn');
  if (!btn) return;
  const sync = (mode) => {
    document.documentElement.dataset.mode = mode;
    btn.textContent = mode === 'dark' ? '☀' : '☾';
    btn.setAttribute('aria-pressed', mode === 'dark' ? 'true' : 'false');
  };
  sync(prefs.get().mode);
  btn.addEventListener('click', async () => {
    const next = prefs.get().mode === 'dark' ? 'light' : 'dark';
    sync(next);
    await prefs.setMode(next);
  });
  window.addEventListener('edu:mode-changed', (e) => sync(e.detail.mode));
  window.addEventListener('storage', (e) => { if (e.key === 'edu:mode') sync(prefs.get().mode); });
}

export function initLang(selectEl) {
  const sel = selectEl || document.getElementById('lang-sel');
  if (!sel) return;
  const apply = (lang) => {
    const l = lang || document.documentElement.lang || 'en';
    sel.value = l;
    document.documentElement.lang = l;
    document.documentElement.dir  = l === 'ar' ? 'rtl' : 'ltr';
  };
  apply(prefs.get().lang);
  sel.addEventListener('change', async () => {
    apply(sel.value);
    await prefs.setLang(sel.value);
  });
  window.addEventListener('edu:lang-changed', (e) => apply(e.detail.lang));
  window.addEventListener('storage', (e) => { if (e.key === 'edu:lang') apply(prefs.get().lang); });
}

export function initUserChip() {
  const chip = document.getElementById('user-chip');
  if (!chip) return;
  auth.onChange((user) => {
    if (user) {
      chip.textContent = user.name || user.email || 'You';
      chip.removeAttribute('data-i18n');
    } else {
      chip.setAttribute('data-i18n', 'nav.signin');
      applyI18n(chip);
    }
    chip.title  = user?.email || '';
    chip.dataset.signedin = user ? '1' : '0';
  });
  chip.addEventListener('click', async () => {
    if (!auth.getUser()) {
      // Testing UI — swap for a real form in Supabase mode
      const email = prompt('Email');    if (!email) return;
      const pw    = prompt('Password'); if (!pw)    return;
      try { await auth.signIn(email, pw); }
      catch (e) { alert(e.message); }
    } else if (confirm('Sign out?')) {
      await auth.signOut();
    }
  });
  // Keep current lang consistent for the chip when language changes while signed out.
  window.addEventListener('edu:lang-changed', () => {
    if (!auth.getUser()) applyI18n(chip);
  });
}

/** Mark cross-app links to open in a new tab so installed windows don't morph. */
export function externalAppLink(appId) {

  if (!a) return;
  a.target = '_blank';
  a.rel = 'noopener noreferrer';
}

// Convenience: bring up all header controls with one call.
export function initHeader() {
  initTheme();
  initLang();
  initUserChip();
}

```js
// shared/install.js — one install prompt per app, plus manual iOS hint.
export function initInstallButton() {
  const btn = document.getElementById('install-btn');
  if (!btn) return;

  const standalone =
    matchMedia('(display-mode: standalone)').matches ||
    matchMedia('(display-mode: minimal-ui)').matches  ||
    navigator.standalone === true;
  if (standalone) { btn.hidden = true; return; }

  let promptEvent = null;

  addEventListener('beforeinstallprompt', (e) => {
    e.preventDefault();
    promptEvent = e;
    btn.hidden = false;
    btn.setAttribute('data-i18n', 'nav.install');
    // trigger i18n for the newly-visible label
    document.dispatchEvent(new CustomEvent('partials:ready'));
  });

  addEventListener('appinstalled', () => {
    btn.hidden = true;
    promptEvent = null;
  });

  const isIos = /iphone|ipad|ipod/i.test(navigator.userAgent);
  if (isIos) {
    btn.hidden = false;
    btn.setAttribute('data-i18n', 'nav.addHome');
    document.dispatchEvent(new CustomEvent('partials:ready'));
  }

  btn.addEventListener('click', async () => {
    if (isIos) {
      alert('Tap the Share button, then "Add to Home Screen".');
      return;
    }
    if (!promptEvent) return;
    promptEvent.prompt();
    try { await promptEvent.userChoice; } catch {}
    promptEvent = null;
    btn.hidden = true;
  });
}
```

## Contract summary (so message 3+ stays consistent)

- **Partials** are declared as `<div data-include="/absolute/path.html"></div>`. `include.js` resolves them before `partials:ready`.
- **Translations** live in `window.I18N` on each page. Keys are page-local. Attributes use `data-i18n-attr="aria-label:key;title:key2"`.
- **Header controls** need three IDs to auto-wire: `#theme-btn`, `#lang-sel`, `#user-chip`. Install button is `#install-btn`. `initHeader()` (or the three separate inits) must be called from each page's entry script.
- **Storage** goes through `prefs.*` and `appData.*` only. No direct `localStorage` calls in page scripts. Namespacing: cross-app `edu:*`, per-app `<appId>:*`.
- **Auth** goes through `auth.*`. Nothing else touches `edu:session` except `auth.js`.

## `sw.js` — v3

```js
/* Edu Dash — single service worker, scope "/". */
const VERSION = 'v3';
const CACHE   = `edu-dash-${VERSION}`;

const SHELL = [
  '/', '/index.html', '/offline.html', '/apps.json', '/register-sw.js',
  '/shared/config.js', '/shared/store.js', '/shared/auth.js',
  '/shared/i18n.js', '/shared/include.js',
  '/shared/ui.js', '/shared/install.js'
];

self.addEventListener('install', (event) => {
  event.waitUntil((async () => {
    const cache = await caches.open(CACHE);
    await Promise.all(SHELL.map((u) => cache.add(u).catch(() => {})));

    // Precache each app's entry page so freshly-installed apps work offline.
    try {
      const res = await fetch('/apps.json', { cache: 'no-cache' });
      const { apps = [] } = await res.json();
      await Promise.all(apps.map((a) => cache.add(a.path).catch(() => {})));
    } catch { /* offline during install — fine */ }

    await self.skipWaiting();
  })());
});

self.addEventListener('activate', (event) => {
  event.waitUntil((async () => {
    const keys = await caches.keys();
    await Promise.all(
      keys.filter((k) => k.startsWith('edu-dash-') && k !== CACHE)
          .map((k) => caches.delete(k))
    );
    await self.clients.claim();
  })());
});

self.addEventListener('message', (e) => {
  if (e.data?.type === 'SKIP_WAITING') self.skipWaiting();
});

/* HTML/CSS/JS/JSON: network-first so deploys land on the next visit.
   Images/fonts/media:        stale-while-revalidate.
   Cross-origin (e.g. supabase.co): bypass — let the browser handle it. */
const NETWORK_FIRST = /\.(?:html?|css|js|mjs|json|webmanifest)$/i;

self.addEventListener('fetch', (event) => {
  const req = event.request;
  if (req.method !== 'GET') return;
  if (req.headers.has('range')) return;

  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return;

  const isNav = req.mode === 'navigate' || url.pathname.endsWith('/');
  if (isNav || NETWORK_FIRST.test(url.pathname)) {
    event.respondWith(networkFirst(req, isNav));
  } else {
    event.respondWith(staleWhileRevalidate(req));
  }
});

async function networkFirst(req, isNav) {
  const cache = await caches.open(CACHE);
  try {
    const fresh = await fetch(req);
    if (fresh.ok) cache.put(req, fresh.clone());
    return fresh;
  } catch {
    const cached =
      (await cache.match(req, { ignoreSearch: true })) ||
      (await cache.match(new URL(req.url).pathname));
    if (cached) return cached;
    if (isNav) {
      const offline = await cache.match('/offline.html');
      if (offline) return offline;
    }
    return Response.error();
  }
}

async function staleWhileRevalidate(req) {
  const cache = await caches.open(CACHE);
  const cached = await cache.match(req);
  const network = fetch(req)
    .then((res) => { if (res.ok) cache.put(req, res.clone()); return res; })
    .catch(() => null);
  return cached || (await network) || Response.error();
}
```

## File contracts (short)

- **Partials**: `<div data-include="/absolute/path.html"></div>` — resolved
  by `shared/include.js` before the `partials:ready` event fires.
- **Translations**: each HTML page sets `window.I18N = { key: {en,fr,ar} }`
  and tags elements with `data-i18n="key"` (or `data-i18n-attr="aria-label:key"`
  for attributes). Missing keys fall back to the element's current text.
- **Storage**: always via `prefs.*` or `appData.*` from `shared/store.js`.
  Namespace: cross-app `edu:*`, per-app `<appId>:*`. Never call
  `localStorage` directly from page scripts.
- **Auth**: only via `shared/auth.js`. `edu:session` is managed by it alone.
- **Header controls**: IDs `#theme-btn`, `#lang-sel`, `#user-chip`,
  `#install-btn` are auto-wired by `initHeader()` / `initInstallButton()`.

## Offline + updates

- HTML / CSS / JS / JSON: **network-first** → new deploys visible on next
  visit; every visited page cached offline.
- Images / fonts: **stale-while-revalidate** → fast, refreshed in background.
- Partials are cached when first fetched (i.e. first time a page using them
  is viewed online). A page that has never been opened online is *not* offline-
  available — its partials don't exist yet. This is intentional; the SW's
  `SHELL` covers the shared bootstrap.
- Service worker takes over immediately (`skipWaiting` + `clients.claim`).
- Bump `VERSION` in `sw.js` **only** when you change SW logic or need to
  purge the whole cache. Content edits do not need a bump.

## Adding nested pages

Drop files anywhere under `<app>/`, e.g. `/arabic/year7/module1/lesson1.html`,
and:
- include `<script src="/register-sw.js" defer></script>`
- include your app header/footer partials and (optionally) module partials
- declare `window.I18N` for the page
- import `shared/ui.js` + `shared/install.js` for the header controls
- put a module-scoped `styles.css` and any specific JS (e.g. canvas drawing)
  next to the page

    npx serve .

Open `http://localhost:3000/arabic/`. Service workers require localhost or
HTTPS. DevTools → Application → Manifest / Service Workers / Cache Storage.

Debug cross-app state: open `/shared/debug.html`.

## Important

- Storage (localStorage / IndexedDB / cookies / Cache) is **shared by origin**
  across all apps. Keys are namespaced (`edu:*`, `arabic:*`, `philo:*`, …).
- Uninstalling one app with "clear data" wipes the origin's data for **all**
  apps; offline pages re-cache on the next online visit.
- Auth in `local` mode stores a session in `localStorage['edu:session']` —
  fine for testing, **never** for real credentials. Switch `CONFIG.backend`
  to `'supabase'` when ready; the schema + wiring are documented at the end

export function attachCanvas(canvas) {
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  const dpr = devicePixelRatio || 1;
  const rect = canvas.getBoundingClientRect();
  canvas.width  = Math.round(rect.width  * dpr);
  canvas.height = Math.round(rect.height * dpr);
  ctx.scale(dpr, dpr);

  let drawing = false;
  const draw = (x, y) => {
    ctx.lineTo(x, y);
    ctx.stroke();
  };
  const pos = (e) => {
    const r = canvas.getBoundingClientRect();
    return [e.clientX - r.left, e.clientY - r.top];
  };
  canvas.addEventListener('pointerdown', (e) => {
    drawing = true;
    ctx.beginPath();
    ctx.moveTo(...pos(e));
    canvas.setPointerCapture(e.pointerId);
  });
  canvas.addEventListener('pointermove', (e) => {
    if (!drawing) return;
    draw(...pos(e));
  });
  const end = () => { drawing = false; };
  canvas.addEventListener('pointerup', end);
  canvas.addEventListener('pointercancel', end);
}
`;

## `arabic/assets/js/app.js`

```js
// arabic/assets/js/app.js — app entry helpers.
// Storage goes only through shared/store.js. Never touch localStorage directly.

import { appData } from '/shared/store.js';

const APP = 'arabic';

/** Increment and return the per-device visit counter for this app. */
export async function trackVisit() {
  const n = ((await appData.get(APP, 'visits', 0)) || 0) + 1;
  await appData.set(APP, 'visits', n);
  return n;
}

/** Read progress object for a module (e.g. 'year7/module1'). */
export async function getProgress(moduleId) {
  return (await appData.get(APP, `progress:${moduleId}`, {})) || {};
}

/** Merge a patch into the module's progress object and persist. */
export async function setProgress(moduleId, patch) {
  const cur  = await getProgress(moduleId);
  const next = { ...cur, ...patch };
  await appData.set(APP, `progress:${moduleId}`, next);
  return next;
}

/** Mark a lesson complete for a module. */
export async function markLesson(moduleId, lessonId, done = true) {
  const cur = await getProgress(moduleId);
  const lessons = { ...(cur.lessons || {}), [lessonId]: !!done };
  return setProgress(moduleId, { lessons });
}

```js
// arabic/year7/module1/draw.js — canvas drawing helper, scoped to the module.
// Imported only by lesson1.html. Kept in module folder (not /assets) on purpose:
// it's specific to this module's UX and should ship with it.

/**
 * Attach pointer drawing to a canvas. Handles HiDPI + resize + pointer capture.
 * @param {HTMLCanvasElement} canvas
 * @param {{ color?: string, width?: number }} [opts]
 * @returns {() => void}  teardown function
 */
export function attachCanvas(canvas, opts = {}) {
  if (!canvas) return () => {};
  const ctx = canvas.getContext('2d');
  const color = opts.color
    || getComputedStyle(document.documentElement)
         .getPropertyValue('--fg').trim()
    || '#1b1f24';
  const width = opts.width || 2.5;

  ctx.lineCap  = 'round';
  ctx.lineJoin = 'round';

  let drawing = false;
  let last = null;

  function resize() {
    const dpr  = window.devicePixelRatio || 1;
    const rect = canvas.getBoundingClientRect();
    if (rect.width === 0 || rect.height === 0) return;

    // snapshot current drawing so resize doesn't erase it
    const prev = document.createElement('canvas');
    prev.width  = canvas.width;
    prev.height = canvas.height;
    if (prev.width && prev.height) {
      prev.getContext('2d').drawImage(canvas, 0, 0);
    }

    canvas.width  = Math.max(1, Math.round(rect.width  * dpr));
    canvas.height = Math.max(1, Math.round(rect.height * dpr));

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.lineCap  = 'round';
    ctx.lineJoin = 'round';

    if (prev.width && prev.height) {
      ctx.drawImage(prev, 0, 0, prev.width / dpr, prev.height / dpr);
    }
  }

  function pos(e) {
    const r = canvas.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top };
  }

  function down(e) {
    drawing = true;
    const p = pos(e);
    last = p;
    ctx.strokeStyle = color;
    ctx.fillStyle   = color;
    ctx.lineWidth   = width;
    // dot at the tap point
    ctx.beginPath();
    ctx.arc(p.x, p.y, width / 2, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(p.x, p.y);
    try { canvas.setPointerCapture(e.pointerId); } catch {}
  }

  function move(e) {
    if (!drawing) return;
    const p = pos(e);
    ctx.beginPath();
    ctx.moveTo(last ? last.x : p.x, last ? last.y : p.y);
    ctx.lineTo(p.x, p.y);
    ctx.stroke();
    last = p;
  }

  function up(e) {
    if (!drawing) return;
    drawing = false;
    last = null;
    if (e && e.pointerId != null) {
      try { canvas.releasePointerCapture(e.pointerId); } catch {}
    }
  }

  canvas.addEventListener('pointerdown', down);
  canvas.addEventListener('pointermove', move);
  canvas.addEventListener('pointerup', up);
  canvas.addEventListener('pointercancel', up);
  canvas.addEventListener('pointerleave', up);

  // Initial size + follow layout changes
  resize();
  const ro = new ResizeObserver(() => resize());
  ro.observe(canvas);

  return function destroy() {
    ro.disconnect();
    canvas.removeEventListener('pointerdown', down);
    canvas.removeEventListener('pointermove', move);
    canvas.removeEventListener('pointerup', up);
    canvas.removeEventListener('pointercancel', up);
    canvas.removeEventListener('pointerleave', up);
  };
}

/** Clear the canvas completely (resets transform too). */
export function clearCanvas(canvas) {
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  ctx.save();
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.restore();
}
```

        const feedback = block.querySelector('[data-feedback]');
        const chosen = block.querySelector(`input[name="${qid}"]:checked`);

        if (chosen && chosen.value === answer) {
          feedback.dataset.state = 'ok';
          feedback.dataset.i18n = 'page.correct';
          applyI18n(feedback);
        } else {
          allCorrect = false;
          feedback.dataset.state = 'ko';
          feedback.dataset.i18n = 'page.incorrect';
          applyI18n(feedback);
        }
      }

      if (allCorrect) {
        await markLesson(MODULE, LESSON, true);
        await renderStatus();
      }
    });

    async function renderStatus() {
      const p = await getProgress(MODULE);
      const done = !!(p.lessons && p.lessons[LESSON]);
      const el = document.getElementById('status');
      if (!el) return;
      el.dataset.i18n = done ? 'page.completed' : 'page.notStarted';
      applyI18n(el);
      el.style.color = done ? 'var(--brand)' : 'var(--muted)';
    }

    renderStatus();
  </script>
</body>
</html>
```

## What this demonstrates

| Concept | Where |
|---|---|
| App partial → module partial → page → module partial → app partial | Both lesson pages |
| Module-scoped `styles.css` | Loaded after app `styles.css`, only on module pages |
| Module-scoped `draw.js` | Lives in module folder, imported only by `lesson1.html` |
| Complete `window.I18N` per page | Includes every key from all partials on that page + page keys |
| `data-i18n-attr` for a11y | `aria-label` on module nav + canvas |
| Dynamic i18n | `#status` flips between `page.completed` / `page.notStarted` via `applyI18n(el)` |
| Storage via `appData` | `markLesson` / `getProgress` write to `arabic:progress:year7/module1` |
| RTL-safe "back" arrow | CSS `[dir="rtl"] .mod-back::before` |
| HiDPI canvas + resize | `draw.js` handles DPR, `ResizeObserver`, pointer capture |
| No localStorage in page code | Only through `shared/store.js` via `arabic/assets/js/app.js` |

## Note on the repeated `window.I18N` blocks

Per the contract, **each page defines its own `window.I18N`** — that's why `app.*`, `nav.*`, `footer.*`, `mod.*` appear in both lessons. If a module grows to dozens of pages and you want to factor common keys out, the clean pattern is a small classic script the page loads **before** its own `window.I18N`:

```html
<script src="/arabic/year7/module1/common-i18n.js"></script>
<script>
  window.I18N = { ...window.I18N, 'page.title': { … } };
</script>
```

`i18n.js` just reads `window.I18N` at runtime, so merging ahead of it is transparent. I did **not** add that file now — it goes against the "complete per page" rule we agreed, and one duplicated set of ~10 keys across 2 pages is fine for the sample. Say the word if you want a `common-i18n.js` convention when the module count grows.

// philo/assets/js/app.js — app entry helpers.
// Storage goes only through shared/store.js. Never touch localStorage directly.

import { appData } from '/shared/store.js';

const APP = 'philo';

/** Increment and return the per-device visit counter for this app. */
export async function trackVisit() {
  const n = ((await appData.get(APP, 'visits', 0)) || 0) + 1;
  await appData.set(APP, 'visits', n);
  return n;
}

/** Read progress object for a module (e.g. 'term1'). */
export async function getProgress(moduleId) {
  return (await appData.get(APP, `progress:${moduleId}`, {})) || {};
}

/** Merge a patch into the module's progress object and persist. */
export async function setProgress(moduleId, patch) {
  const cur  = await getProgress(moduleId);
  const next = { ...cur, ...patch };
  await appData.set(APP, `progress:${moduleId}`, next);
  return next;
}

/** Mark a lesson complete for a module. */
export async function markLesson(moduleId, lessonId, done = true) {
  const cur = await getProgress(moduleId);
  const lessons = { ...(cur.lessons || {}), [lessonId]: !!done };
  return setProgress(moduleId, { lessons });
}

/** Draft text (persisted per lesson). */
export async function getDraft(moduleId, lessonId) {
  return (await appData.get(APP, `draft:${moduleId}:${lessonId}`, '')) || '';
}
export async function setDraft(moduleId, lessonId, text) {
  return appData.set(APP, `draft:${moduleId}:${lessonId}`, text || '');
}
```

## `philo/term1/writing.js`

```js
// philo/term1/writing.js — text autosave helper, scoped to the module.
// Different shape from arabic's draw.js on purpose: it shows the pattern
// generalizes — module-specific JS can be anything the module needs.

/**
 * Autosave textarea content after a short idle period.
 * @param {HTMLTextAreaElement} textarea
 * @param {{
 *   delay?: number,
 *   onSave: (value: string) => Promise<void>,
 *   onState?: (state: 'idle'|'saving'|'saved') => void
 * }} opts
 * @returns {() => void}  teardown
 */
export function attachAutosave(textarea, { delay = 600, onSave, onState } = {}) {
  if (!textarea || typeof onSave !== 'function') return () => {};

  let timer = null;
  let lastSaved = textarea.value;
  let saving = false;
  let pending = false;

  function setState(s) { if (onState) onState(s); }

  async function flush() {
    if (saving) { pending = true; return; }
    const value = textarea.value;
    if (value === lastSaved) { setState('idle'); return; }
    saving = true;
    setState('saving');
    try {
      await onSave(value);
      lastSaved = value;
      setState('saved');
    } catch (err) {
      console.error('[autosave]', err);
      setState('idle');
    } finally {
      saving = false;
      if (pending) { pending = false; await flush(); }
    }
  }

  function scheduleSave() {
    clearTimeout(timer);
    timer = setTimeout(flush, delay);
  }

  textarea.addEventListener('input', scheduleSave);

  // Save on navigate away (visibility change catches mobile tab switch too)
  function flushOnHide() {
    if (document.visibilityState === 'hidden') flush();
  }
  document.addEventListener('visibilitychange', flushOnHide);

  return function destroy() {
    clearTimeout(timer);
    textarea.removeEventListener('input', scheduleSave);
    document.removeEventListener('visibilitychange', flushOnHide);
  };
}
```

## `docs/SUPABASE.md`

```md
# Supabase migration — flipping `backend: 'supabase'`

## Why bother

`localStorage` on a shared origin:

- Is wiped for **all** apps when the user uninstalls **any** one of them.
- Has no per-user isolation.
- Cannot safely hold credentials.

Supabase fixes the first two. It also gives you real auth (email/password,
magic link, OAuth) without any server code.

## What actually changes

Three source files and one SQL script. **No changes to `arabic/` or `philo/`.**

| File | Change |
|---|---|
| `shared/config.js` | flip `backend: 'supabase'`, fill `supabaseUrl`, `supabaseAnonKey` |
| `shared/store.js`  | replace the `supabase` stub with the body below |
| `shared/auth.js`   | replace the `supabaseAuth` stub with the body below |
| `sw.js`            | bump `VERSION` (optional but recommended — data shape changed) |
| Supabase project   | run the SQL below |

The facade in `store.js` / `auth.js` exposes the **same public API** as the
local backend, so app pages (`arabic/`, `philo/`) keep working unmodified.

---

## 1. Create the Supabase project

1. supabase.com → New project (free tier is enough).
2. Once ready: Project Settings → API → copy **Project URL** and **anon public key**.
3. Save them for step 3 below.

> The `anon` key is *public* by design. Row Level Security (RLS) is what
> actually protects data — see the policies in step 2.

---

## 2. Schema + RLS — paste into Supabase SQL Editor and run

```sql
-- One row per user: shared prefs live here.
create table profiles (
  id         uuid primary key references auth.users(id) on delete cascade,
  name       text,
  avatar_url text,
  prefs      jsonb not null default '{}'::jsonb,   -- { mode, lang }
  updated_at timestamptz not null default now()
);

-- Per-app key/value. The composite PK is what isolates apps.
create table app_data (
  user_id    uuid not null references auth.users(id) on delete cascade,
  app_id     text not null,
  key        text not null,
  value      jsonb,
  updated_at timestamptz not null default now(),
  primary key (user_id, app_id, key)
);

create index app_data_user_app_idx on app_data(user_id, app_id);

alter table profiles  enable row level security;
alter table app_data  enable row level security;

create policy "profiles: own row" on profiles
  for all
  using  (auth.uid() = id)
  with check (auth.uid() = id);

create policy "app_data: own rows" on app_data
  for all
  using  (auth.uid() = user_id)
  with check (auth.uid() = user_id);
```

**Why this is enough:**

- Every read/write from the client carries the user's JWT. Postgres checks
  `auth.uid()` against the row's `id` / `user_id`. Cross-user leaks are
  impossible without a stolen JWT.
- `app_id` is client-supplied but constrained by the RLS policy to the
  authenticated user's own rows — one user can't see another user's data.
- **Uninstalling** the Arabic PWA does **not** affect `app_data` rows for
  Philo, because those are rows on the server, keyed by `(user_id, app_id)`.

## 4. `shared/store.js` — replace the `supabase` stub

Replace the whole `const supabase = { … }` object with:

```js
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const sb = createClient(CONFIG.supabaseUrl, CONFIG.supabaseAnonKey, {
  auth: { persistSession: true, autoRefreshToken: true }
});
export { sb };   // auth.js imports it below

const supabase = {
  async pushPrefs() {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return;
    const prefs = {
      mode: mirror.read(PREF_KEYS.mode, null),
      lang: mirror.read(PREF_KEYS.lang, null)
    };
    await sb.from('profiles').upsert({
      id: user.id, prefs, updated_at: new Date().toISOString()
    });
  },

  async pullPrefs() {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return null;
    const { data } = await sb.from('profiles')
      .select('prefs, name, avatar_url').eq('id', user.id).maybeSingle();
    if (!data) return null;
    return {
      mode: data.prefs?.mode,
      lang: data.prefs?.lang,
      profile: data.name ? { name: data.name, avatar: data.avatar_url } : null
    };
  },

  async getApp(appId, key) {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return null;
    const { data } = await sb.from('app_data')
      .select('value').eq('user_id', user.id)
      .eq('app_id', appId).eq('key', key).maybeSingle();
    return data?.value ?? null;
  },

  async setApp(appId, key, val) {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return;
    await sb.from('app_data').upsert({
      user_id: user.id, app_id: appId, key, value: val,
      updated_at: new Date().toISOString()
    });
  },

  async removeApp(appId, key) {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return;
    await sb.from('app_data').delete()
      .eq('user_id', user.id).eq('app_id', appId).eq('key', key);
  },

  async clearApp(appId) {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return;
    await sb.from('app_data').delete()
      .eq('user_id', user.id).eq('app_id', appId);
  },

  async setProfile(p) {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) return;
    await sb.from('profiles').upsert({
      id: user.id,
      name: p?.name ?? null,
      avatar_url: p?.avatar ?? null
    });
  }
};
```

> **Offline note:** writes still land in the local mirror first
> (`mirror.write(...)` inside `appData.set`), then `await backend.setApp(...)`.
> If the network is down, the Supabase call rejects — `appData.set` currently
> `await`s it, so the caller sees the error. Two options:
> (a) leave as-is (UI blocks / retries when back online), or
> (b) wrap backend calls in try/catch + enqueue to an outbox. For v1,
> the simplest correct choice is to catch in `appData` and log:
>
> ```js
> async set(appId, key, val) {
>   mirror.write(`${appId}:${key}`, val);
>   try { await backend.setApp(appId, key, val); }
>   catch (e) { console.warn('[appData.set] offline:', e); }
> }
> ```

## 5. `shared/auth.js` — replace the `supabaseAuth` stub

```js
import { sb } from './store.js';       // reuses the same Supabase client
import { prefs } from './store.js';

const supabaseAuth = {
  async signIn(email, password) {
    const { data, error } = await sb.auth.signInWithPassword({ email, password });
    if (error) throw error;
    await prefs.sync();                     // pull shared prefs from server
    notify();
    return data.user;
  },

  async signUp(email, password, name) {
    const { data, error } = await sb.auth.signUp({
      email, password,
      options: { data: { name } }
    });
    if (error) throw error;
    if (data.user) {
      await sb.from('profiles').upsert({
        id: data.user.id, name: name || email.split('@')[0]
      });
    }
    notify();
    return data.user;
  },

  async signOut() {
    await sb.auth.signOut();
    localStorage.removeItem(SESSION_KEY);
    await prefs.setProfile(null);
    notify();
  },

  getUser() {
    try { return JSON.parse(localStorage.getItem(SESSION_KEY)); }
    catch { return null; }
  }
};

// Mirror Supabase session into our local mirror for the sync chip.
// (Supabase SDK itself stores the session; we only copy the display bits.)
sb.auth.onAuthStateChange((_evt, session) => {
  if (session?.user) {
    localStorage.setItem(SESSION_KEY, JSON.stringify({
      id:    session.user.id,
      email: session.user.email,
      name:  session.user.user_metadata?.name
    }));
  } else {
    localStorage.removeItem(SESSION_KEY);
  }
  notify();
});
```

> ⚠️ `notify()` in `auth.js` already exists (it fires `onChange` listeners).
> Keep it as-is. The chip in `shared/ui.js` reacts automatically.
>
> **First sign-in after switching backends:** the user has local data from
> `local` mode; the server has none. Either (a) clear local data on first
> Supabase sign-in (`prefs.sync()` returning `null` means server empty), or
> (b) accept that the local mirror is empty until they re-enter settings.
> Option (a) is cleaner; add to `auth.signIn`:
>
> ```js
> const remote = await backend.pullPrefs();
> if (!remote) {
>   // seed the server from whatever is still in local mirror
>   await backend.pushPrefs();
>   await backend.setProfile(prefs.get().profile);
> }
> ```

## 8. Rollback

Flip `CONFIG.backend` back to `'local'`. The app uses `localStorage` again.
Remote data is untouched. **There is no automatic merge** back into local
storage — if you roll back and want to keep data, export from Supabase
first (`Table Editor → app_data → Download CSV`).

---

## 9. What this does *not* solve

- **Offline writes**: Supabase is HTTP. Writes fail offline. Add an outbox
  (IndexedDB + Background Sync) later if that matters. Reads still work from
  the local mirror, so the UI stays populated.
- **Passwords in `localStorage`**: never happened. Supabase SDK stores the
  *session* (JWT + refresh token) in `localStorage`, which is standard and
  reasonable — it's not a password, and Supabase rotates it.
- **Cross-tab sign-in**: works out of the box via `onAuthStateChange`. The
  local mirror's `storage` event listener also fires `notify()` if another
  tab writes it, so the chip updates in every tab.
